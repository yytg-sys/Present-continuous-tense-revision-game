        const AudioContext = window.AudioContext || window.webkitAudioContext;
        let audioCtx = null;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new AudioContext();
            }
        }

        function playSound(type) {
            if (!audioCtx) return;
            try {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                const now = audioCtx.currentTime;

                if (type === 'correct') {
                    // Happy coin chime
                    osc.type = 'square';
                    osc.frequency.setValueAtTime(523.25, now); // C5
                    osc.frequency.setValueAtTime(659.25, now + 0.08); // E5
                    osc.frequency.setValueAtTime(783.99, now + 0.16); // G5
                    osc.frequency.setValueAtTime(1046.50, now + 0.24); // C6
                    gain.gain.setValueAtTime(0.15, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
                    osc.start(now);
                    osc.stop(now + 0.4);
                } else if (type === 'wrong') {
                    // Sad buzz
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(220, now); // A3
                    osc.frequency.setValueAtTime(164.81, now + 0.15); // E3
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.35);
                    osc.start(now);
                    osc.stop(now + 0.35);
                } else if (type === 'win') {
                    // Fanfare
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(440, now);
                    osc.frequency.setValueAtTime(554.37, now + 0.12);
                    osc.frequency.setValueAtTime(659.25, now + 0.24);
                    osc.frequency.setValueAtTime(880, now + 0.36);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.7);
                    osc.start(now);
                    osc.stop(now + 0.7);
                }
            } catch (e) {
                console.log("Audio play error:", e);
            }
        }

        // Pixel Rabbit Drawing Function
        function drawRabbit(x, y, frame, state) {
            ctx.save();
            ctx.translate(x, y);

            // Shadow
            ctx.fillStyle = 'rgba(0, 0, 0, 0.25)';
            ctx.beginPath();
            ctx.ellipse(16, 30, 14, 5, 0, 0, Math.PI * 2);
            ctx.fill();

            // Rabbit Body Frame bounce
            const bounceY = (state === 'running') ? (frame % 2 === 0 ? 0 : -3) : 0;
            ctx.translate(0, bounceY);

            // Feet movement when running
            ctx.fillStyle = '#e2e8f0'; // Light slate / white body
            if (state === 'running') {
                const legOffset = (frame % 2 === 0) ? 4 : -4;
                ctx.fillRect(4 + legOffset, 24, 8, 6);
                ctx.fillRect(18 - legOffset, 24, 8, 6);
            } else {
                ctx.fillRect(6, 24, 8, 6);
                ctx.fillRect(18, 24, 8, 6);
            }

        // Render Grassland Environment Loop
        function renderScene() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const w = canvas.width;
            const h = canvas.height;
            const horizon = h * 0.65;

            // Sky Gradient
            const skyGrad = ctx.createLinearGradient(0, 0, 0, horizon);
            skyGrad.addColorStop(0, '#38bdf8');
            skyGrad.addColorStop(1, '#bae6fd');
            ctx.fillStyle = skyGrad;
            ctx.fillRect(0, 0, w, horizon);

            // 8-bit Clouds
            ctx.fillStyle = '#ffffff';
            const cloudX1 = (bgOffset * 0.2) % (w + 120) - 60;
            const cloudX2 = (bgOffset * 0.1 + 200) % (w + 160) - 80;
            
            // Cloud 1
            ctx.fillRect(w - cloudX1, 30, 50, 16);
            ctx.fillRect(w - cloudX1 + 10, 20, 30, 10);
            
            // Cloud 2
            ctx.fillRect(w - cloudX2, 60, 60, 20);
            ctx.fillRect(w - cloudX2 + 15, 48, 30, 14);

            // Distant Hills
            ctx.fillStyle = '#16a34a';
            ctx.beginPath();
            ctx.moveTo(0, horizon);
            for (let i = 0; i <= w; i += 40) {
                const hillY = horizon - Math.sin((i + bgOffset * 0.5) * 0.01) * 20;
                ctx.lineTo(i, hillY);
            }
            ctx.lineTo(w, horizon);
            ctx.fill();

            // Grassland Floor
            ctx.fillStyle = '#22c55e';
            ctx.fillRect(0, horizon, w, h - horizon);

            // Grass Dirt Border Line
            ctx.fillStyle = '#15803d';
            ctx.fillRect(0, horizon, w, 6);

            // Moving Grass Blades & Details
            ctx.fillStyle = '#166534';
            const step = 40;
            const offsetGrass = bgOffset % step;
            for (let x = -offsetGrass; x < w; x += step) {
                ctx.fillRect(x, horizon + 12, 6, 8);
                ctx.fillRect(x + 18, horizon + 30, 8, 6);
                ctx.fillRect(x + 28, horizon + 50, 6, 10);
            }

            // Draw Floating Target Carrot if active
            if (showCarrotInWorld) {
                drawCarrot(carrotX, carrotY);
            }

            // Draw Rabbit
            const rabbitX = w * 0.2;
            const rabbitY = horizon - 20;
            const rabbitState = (isRunning && !gameFinished) ? 'running' : 'idle';
            drawRabbit(rabbitX, rabbitY, Math.floor(frameCount / 8), rabbitState);

            // If Game Win with Full Energy -> Show Big Reward Carrot
            if (gameFinished && lives > 0 && energy >= 99) {
                drawBigCarrot(w * 0.55, horizon - 50);
            }
        }

        // Main Animation Loop
        function gameLoop() {
            frameCount++;
            if (isRunning) {
                bgOffset += 4;
                if (showCarrotInWorld) {
                    carrotX -= 4;
                    if (carrotX < canvas.width * 0.2 + 10) {
                        showCarrotInWorld = false; // Rabbit "eats" carrot
                    }
                }
            }

            renderScene();
            requestAnimationFrame(gameLoop);
        }

            if (answeredCurrent || gameFinished) return;
            answeredCurrent = true;

            const q = questions[currentQuestionIdx];
            const feedback = document.getElementById('feedback-banner');

            if (selectedIndex === q.correct) {
                // CORRECT ANSWER: Spawn carrot for rabbit to eat!
                playSound('correct');
                
                carrotX = canvas.width - 60;
                carrotY = canvas.height * 0.65 - 30;
                showCarrotInWorld = true;

                // Resume 1/3 of max energy (approx +33.33%), max cap at 100
                energy = Math.min(100, energy + 33.33);

                feedback.innerText = `✨ EXCELLENT! ${q.rule}`;
                feedback.className = "absolute bottom-4 left-1/2 -translate-x-1/2 px-4 py-2 rounded-lg border-2 border-black text-xs sm:text-sm font-bold text-center pointer-events-none bg-green-500 text-slate-950 opacity-100 transition-opacity duration-300 z-20";

            } else {
                // WRONG ANSWER: No carrot appears!
                playSound('wrong');
                lives--;
                showCarrotInWorld = false;

                // Deduct energy corresponding to 1/3 lost
                energy = Math.max(0, energy - 33.33);

                feedback.innerText = `❌ OOPS! ${q.rule}`;
                feedback.className = "absolute bottom-4 left-1/2 -translate-x-1/2 px-4 py-2 rounded-lg border-2 border-black text-xs sm:text-sm font-bold text-center pointer-events-none bg-red-500 text-white opacity-100 transition-opacity duration-300 z-20";
            }

            updateHUD();

            // Check Game Over Condition (3 wrong answers / 0 lives / 0 energy)
            if (lives <= 0 || energy <= 0) {
                setTimeout(() => {
                    feedback.style.opacity = '0';
                    triggerGameOver();
                }, 1200);
                return;
            }

            // Proceed to next question after short delay
            setTimeout(() => {
                feedback.style.opacity = '0';
                currentQuestionIdx++;
                showNextQuestion();
            }, 1500);
        }

        function triggerGameOver() {
            gameFinished = true;
            isRunning = false; // Rabbit stops running
            
            document.getElementById('question-overlay').classList.add('hidden');
            
            const resultModal = document.getElementById('result-modal');
            document.getElementById('result-icon').innerText = '🪫';
            document.getElementById('result-title').innerText = 'ENERGY DRAINED!';
            document.getElementById('result-title').className = 'text-base sm:text-lg text-red-400 mb-2';
            document.getElementById('result-message').innerText = 'You can get the surprise gift next time! Try again!';

            resultModal.classList.remove('hidden');
        }

        function triggerWinGame() {
            gameFinished = true;
            document.getElementById('question-overlay').classList.add('hidden');
            playSound('win');

            const resultModal = document.getElementById('result-modal');

            if (energy >= 99) {
                // FINISHED WITH FULL ENERGY!
                document.getElementById('result-icon').innerText = '🥕🏆';
                document.getElementById('result-title').innerText = 'SUPER RABBIT CHAMPION!';
                document.getElementById('result-title').className = 'text-base sm:text-lg text-yellow-400 mb-2';
                document.getElementById('result-message').innerHTML = 'AMAZING! You finished with <strong>FULL ENERGY</strong>!<br><br>🌟 You unlocked the <strong>EXTRA LARGE SURPRISE CARROT</strong> for Bun-Bun! 🎉';
            } else {
                // FINISHED BUT NOT FULL ENERGY
                document.getElementById('result-icon').innerText = '🐰👍';
                document.getElementById('result-title').innerText = 'GREAT JOB!';
                document.getElementById('result-title').className = 'text-base sm:text-lg text-green-400 mb-2';
                document.getElementById('result-message').innerText = 'You helped Bun-Bun finish the run! You can get the surprise gift next time! Try again for 100% full energy!';
            }

            resultModal.classList.remove('hidden');
        }

        function restartGame() {
            document.getElementById('result-modal').classList.add('hidden');
            resetGameState();
            isRunning = true;
            showNextQuestion();
        }

        // Initialize canvas render on load
        window.onload = function() {
            resizeCanvas();
            requestAnimationFrame(gameLoop);
        };
    </script>
